# A punch on the Adam's apple: he grabs his throat for Throat Duration, then fights on.
# Refused: a punch on the chest (outside both zones), a slash on the throat (Blunt Only),
# a punch from behind (Max Hit Angle). The contact point is the neck part moved 5 cm up and
# 5 cm forward, the throat zone's center with the default offsets (BasHarness hit-npc at=).
wait-level
set-option ThroatBalls "Throat Balls Enabled" true
set-option ThroatBalls "Debug Logging" true
set-option ThroatBalls "Own Sounds" true
set-option ThroatBalls "Throat Sound" Random
set-option ThroatBalls "Punch Sound" true
set-option ThroatBalls "Heavy Punch Speed" 4.5
# Repeats every second: the two throat files take turns, so both play during the first reaction.
set-option ThroatBalls "Groan Interval" 1
# Chances pinned: a run must not depend on a roll.
set-option ThroatBalls "Throat Reaction Chance" 1
set-option ThroatBalls "Groin Reaction Chance" 1
set-option ThroatBalls "Groin Knockdown Chance" 0
set-option ThroatBalls "Both Hits Knockdown Chance" 1
set-option ThroatBalls "Allowed Hits" "Fists, Kicks, Blunt Weapons"
set-option ThroatBalls "Metal Armor Protects" false
set-option ThroatBalls "Men Only" true
set-option ThroatBalls "Throat Duration" 3
set-option ThroatBalls "Both Hits Take Him Down" true
frame-rate 90

spawn-npc HumanMale distance=1.5 facing=0 brain=HumanDummy
record-start throat fps=30 width=960 height=540 views=npc distance=1.6 rise=1.3 angle=30
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5
expect-log [ThroatBalls] LOW BLOW
expect-log | Throat start
# Own Sounds (on by default): an own throat file plays at the start, not a game clip. The mod ships
# throat-choke.wav, the harness adds throat-test.wav.
expect-log | own Throat 'throat-
expect-no-log | game Throat
# Punch Sound (on by default): one of the game's punches comes with the first choke. 5 m/s is from
# Heavy Punch Speed (4.5) on: a heavy one.
expect-log | game punch 'SFX_punch_intens2_
expect-log | heavy, hit 5.0 m/s
wait 0.8
screenshot throat-hands of=npc distance=1.6 height=1.5 angle=0
await-log 5 | Throat end (time up)
# The repeats took turns: the shipped choke played in the game, and so did the test tone.
expect-log | own Throat 'throat-choke.wav'
expect-log | own Throat 'throat-test.wav'
wait 1
record-stop
# The brain wakes after the stun clip's end; a HumanDummy must stay without its behaviour tree.
wait 2
expect-npc tree False within=0.1

mark
hit-npc Torso section=Upper at=0,0,0.15 type=Blunt velocity=5
expect-log | OutsideZones
mark
hit-npc Neck at=0,0.05,0.05 type=Slash velocity=5
expect-log | HitNotAllowed
expect-no-log | Throat start

# Own Sounds off: one of the game's four breathless-death clips (Throat Sound Random), not the voice's line.
set-option ThroatBalls "Own Sounds" false
spawn-npc HumanMale distance=1.5 angle=40 facing=0 brain=HumanDummy
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5
expect-log | Throat start
expect-log | game Throat 'Male_Shout-of-Pain_0
expect-no-log | own Throat
expect-no-log | breathlessdeath
expect-no-log | grappled

# Throat Sound 2: always the second clip (Male_Shout-of-Pain_054), also on the repeat (Groan Interval 1 s).
set-option ThroatBalls "Throat Sound" "Sound 2"
expect-option ThroatBalls "Throat Sound" "Sound 2"
spawn-npc HumanMale distance=1.5 angle=-40 facing=0 brain=HumanDummy
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5
expect-log | Throat start
expect-log | game Throat 'Male_Shout-of-Pain_054'
expect-log | game punch 'SFX_punch_intens2_
# The whole reaction (Throat Duration 3 s): the repeats too. The NPC before also took Sound 2 for his last repeats.
# The repeats choke without a punch: nobody hit him again.
mark
wait 3.5
expect-log | game Throat 'Male_Shout-of-Pain_054'
expect-no-log | game Throat 'Male_Shout-of-Pain_047'
expect-no-log | game Throat 'Male_Shout-of-Pain_072'
expect-no-log | game Throat 'Male_Shout-of-Pain_087'
expect-no-log | game punch

# Punch Sound off: the choke alone.
set-option ThroatBalls "Punch Sound" false
expect-option ThroatBalls "Punch Sound" false
spawn-npc HumanMale distance=1.5 angle=80 facing=0 brain=HumanDummy
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5
expect-log | Throat start
expect-log | game Throat 'Male_Shout-of-Pain_054'
expect-no-log | game punch

# A lighter hit (3 m/s, under Heavy Punch Speed): a light punch.
set-option ThroatBalls "Punch Sound" true
spawn-npc HumanMale distance=1.5 angle=-80 facing=0 brain=HumanDummy
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=3
expect-log | Throat start
expect-log | game punch 'SFX_punch_intens1_
expect-log | light, hit 3.0 m/s
expect-no-log | game punch 'SFX_punch_intens2_
# Options are saved in the run's prefix: give them back for the next scenarios.
set-option ThroatBalls "Own Sounds" true
set-option ThroatBalls "Throat Sound" Random
set-option ThroatBalls "Groan Interval" 3
set-option ThroatBalls "Punch Sound" true
expect-no-log Exception
quit
