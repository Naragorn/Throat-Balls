# Films with sound, for listening and looking: every reaction once, on the same man, with the mod as it ships
# (tools/harness.sh --as-shipped harness/films/sound.txt). Not one of the regression runs (tools/harness.sh runs
# harness/*.txt only). The hits come through the game's damage code without a physics contact, so the game's own hit
# sound is missing; what the films have is the mod's sound. Chances are pinned per film, the rest are the defaults
# (Throat Duration 3 s, Groin Duration 4 s, Groan Interval 3 s, Time Between Hits 4 s, Down Duration 8 s).
# Between two films the man gets 5 s, so a hit never counts as the second of both hits by accident.
wait-level
set-option ThroatBalls "Throat Balls Enabled" true
set-option ThroatBalls "Debug Logging" true
set-option ThroatBalls "Throat Reaction Chance" 1
set-option ThroatBalls "Groin Reaction Chance" 1
set-option ThroatBalls "Groin Knockdown Chance" 0
set-option ThroatBalls "Both Hits Knockdown Chance" 1
set-option ThroatBalls "Metal Armor Protects" false
set-option ThroatBalls "Fall Style" CurlUp
frame-rate 90

spawn-npc HumanMale distance=1.0 facing=0 brain=HumanDummy
wait 1

# 1. Throat, a hard hit (5 m/s, from Heavy Punch Speed 4.5 on): the shipped choke with a heavy punch.
record-start throat fps=30 width=960 height=540 views=side,npc distance=1.6 rise=1.3 angle=30 audio=true
wait 0.8
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5 swing=5
expect-log | Throat start
expect-log | own Throat 'throat-choke.wav'
expect-log | heavy, hit 5.0 m/s
await-log 6 | Throat end (time up)
wait 1
record-stop
wait 1

# 2. Groin alone, no knockdown: the grunt, then heavy breathing.
record-start groin fps=30 width=960 height=540 views=side,npc distance=1.8 rise=1.0 angle=60 audio=true
wait 0.8
mark
hit-npc Torso section=Lower at=0,-0.08,0.08 type=Blunt velocity=5 swing=5
expect-log | Groin start
expect-log | game Groin '
expect-no-log | Down start
await-log 7 | Groin end (time up)
wait 1
record-stop
wait 1

# 3. Throat, a lighter hit (3 m/s): the choke with a light punch.
record-start throat-light fps=30 width=960 height=540 views=side,npc distance=1.6 rise=1.3 angle=30 audio=true
wait 0.8
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=3 swing=3
expect-log | Throat start
expect-log | light, hit 3.0 m/s
await-log 6 | Throat end (time up)
wait 1
record-stop
wait 1

# 4. Throat with Own Sounds off: one of the game's four breathless-death clips instead of the shipped choke.
set-option ThroatBalls "Own Sounds" false
record-start throat-game-voice fps=30 width=960 height=540 views=side,npc distance=1.6 rise=1.3 angle=30 audio=true
wait 0.8
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5 swing=5
expect-log | Throat start
expect-log | game Throat 'Male_Shout-of-Pain_0
await-log 6 | Throat end (time up)
wait 1
record-stop
set-option ThroatBalls "Own Sounds" true
wait 1

# 5. Groin alone takes him down (Groin Knockdown Chance 100 % here, 50 % by default): Curl Up, up after 8 s.
set-option ThroatBalls "Groin Knockdown Chance" 1
record-start groin-down fps=30 width=960 height=540 views=side,npc distance=2.2 rise=1.0 angle=60 audio=true
wait 0.8
mark
hit-npc Torso section=Lower at=0,-0.08,0.08 type=Blunt velocity=5 swing=5
expect-log | Down start (groin
await-log 11 | Down end (time up)
wait 3
record-stop
set-option ThroatBalls "Groin Knockdown Chance" 0
expect-npc state Alive within=5
wait 1

# 6. Throat, then the groin 1.5 s later: both hits, he goes down (Curl Up) and gets up after 8 s.
record-start throat-then-groin fps=30 width=960 height=540 views=side,npc distance=2.2 rise=1.0 angle=60 audio=true
wait 0.8
mark
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5 swing=5
expect-log | Throat start
wait 1.5
hit-npc Torso section=Lower at=0,-0.08,0.08 type=Blunt velocity=5 swing=5
expect-log | Down start (both hits
await-log 11 | Down end (time up)
wait 3
record-stop
expect-npc state Alive within=5
wait 1

# 7. Groin, then the throat 1.5 s later, with Fall Style Ragdoll: he drops limp and gets up by himself.
set-option ThroatBalls "Fall Style" Ragdoll
record-start groin-then-throat-ragdoll fps=30 width=960 height=540 views=side,npc distance=2.4 rise=1.0 angle=60 audio=true
wait 0.8
mark
hit-npc Torso section=Lower at=0,-0.08,0.08 type=Blunt velocity=5 swing=5
expect-log | Groin start
wait 1.5
hit-npc Neck at=0,0.05,0.05 type=Blunt velocity=5 swing=5
expect-log | Down start (both hits
await-log 11 | Down end (time up)
expect-npc state Alive within=20
# State Alive comes before he is on his feet; the film runs on until he stands.
wait 4
record-stop
set-option ThroatBalls "Fall Style" CurlUp
expect-no-log Exception
